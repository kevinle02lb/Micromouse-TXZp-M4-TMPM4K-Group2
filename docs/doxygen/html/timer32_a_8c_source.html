313void T32A1_Interrupt_Enable(void)
314{
315 // Fire interrupt on T32AxRGC1. Disable T32AxRGC0
316 TSB_T32A1->IMC &= ~T32A_IMx_IMx1; /* 0 = unmask (Enabled), 1 = mask (Disabled)*/
317 TSB_T32A1->IMC |= T32A_IMx_IMx0;
318
319 /* [1] Disabling Interrupt by CPU */
320 __disable_irq(); /* Disable Interrupt via PRIMASK */
321 /* [2] CPU Registers Setting */
322 __NVIC_SetPriority(INTT32A01AC_IRQn, T32A1_IRQ_PRIORITY); /* Set Priority - Interrupt Priority Register (8Bit wide) */
323 __NVIC_ClearPendingIRQ(INTT32A01AC_IRQn); /* Clear Pending IRQ - Interrupt Clear Pending Register */
324
325 /* [3-5] Ignored */
326
327 /* [7] Enabling Interrupt by CPU */
328 __NVIC_EnableIRQ(INTT32A01AC_IRQn); /* Enables a device specific interrupt in the NVIC interrupt controller */
329 __enable_irq();
330}

84static uint16_t Duty_ToCompare(uint8_t duty, uint16_t period)
85{
86 if (duty >= MAX_DUTY)
87 {
88 return (uint16_t)(period - 1U); /* Near-full drive (1 count brake) */
89 }
90
91 return (uint16_t)((uint32_t)duty * (uint32_t)(period - 1U) / (uint32_t)MAX_DUTY);
92}

132/* ==========================================================================
133 * Left motor (T32A0: IN1 = A/OUTA, IN2 = B/OUTB)
134 * ========================================================================== */
135
136void Motor_SetLeft(motor_dir_t dir, uint8_t speed)
137{
138 uint16_t duty = Duty_ToCompare(speed, PERIOD_L);
139
140 /* Fast path: same direction, update the PWM leg's duty only. FORWARD drives the PWM on IN1 (A); REVERSE on IN2 (B). */
141 if (dir == motor_left_dir)
142 {
143 if (dir == FORWARD)
144 {
145 T32A0_SetTimerA0(duty); /* IN1 is the PWM leg in FORWARD */
146 }
147 else if (dir == REVERSE)
148 {
149 T32A0_SetTimerB0(duty); /* IN2 is the PWM leg in REVERSE */
150 }
151 /* BRAKE / STOP: no live update required */
152 return;
153 }
154
155 /* Full path: stop, reconfigure output control, restart */
156 T32A0_Stop();
157
158 switch (dir)
159 {
160 /* Slow-decay forward: IN1 PWM, IN2 static HIGH (PWM low = drive) */
161 case FORWARD:
162 T32A0_SetOutCRA1(T32A_OUTPUT_PPG); /* IN1 = PWM */
163 T32A0_SetOutCRB1(T32A_OUTPUT_HIGH); /* IN2 = static HIGH */
164 T32A0_SetTimerA0(duty);
165 T32A0_SetTimerB0(0U);
166 break;
167
168 /* Slow-decay reverse: IN1 static HIGH, IN2 PWM (PWM low = drive) */
169 case REVERSE:
170 T32A0_SetOutCRA1(T32A_OUTPUT_HIGH); /* IN1 = static HIGH */
171 T32A0_SetOutCRB1(T32A_OUTPUT_PPG); /* IN2 = PWM */
172 T32A0_SetTimerA0(0U);
173 T32A0_SetTimerB0(duty);
174 break;
175
176 /* IN1=HIGH, IN2=HIGH -> both outputs low -> motor shorted */
177 case BRAKE:
178 T32A0_SetOutCRA1(T32A_OUTPUT_HIGH);
179 T32A0_SetOutCRB1(T32A_OUTPUT_HIGH);
180 T32A0_SetTimerA0(0U);
181 T32A0_SetTimerB0(0U);
182 break;
183
184 /* IN1=LOW, IN2=LOW -> Hi-Z -> standby */
185 case STOP:
186 T32A0_SetOutCRA1(T32A_OUTPUT_LOW);
187 T32A0_SetOutCRB1(T32A_OUTPUT_LOW);
188 T32A0_SetTimerA0(0U);
189 T32A0_SetTimerB0(0U);
190 break;
191 }
192
193 motor_left_dir = dir;
194 T32A0_Start();
195}

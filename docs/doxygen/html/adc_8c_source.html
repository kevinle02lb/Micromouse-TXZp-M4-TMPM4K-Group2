56/* ==========================================================================
57 * INTERRUPT NOTES (Read this first)
58 * ==========================================================================
59 *
60 * ADC single-conversion interrupts:
61 * INTADASGL = IRQ #44 (ADC Unit A done)
62 * INTADCSGL = IRQ #58 (ADC Unit C done)
63 *
64 * These are "direct" interrupts (Route H in Toshiba §4.3.1).
65 * They bypass INTIF and go straight to the NVIC.
66 *
67 * To enable an interrupt manually (without CMSIS):
68 * NVIC_ISER[IRQn >> 5] = (1UL << (IRQn & 0x1F));
69 *
70 * NVIC base address: 0xE000E100 (ARM standard, not in Toshiba doc)
71 * ISER offset: +0x000, ICER: +0x080, ICPR: +0x180, IPR: +0x300
72 *
73 * To enable with CMSIS (recommended):
74 * __NVIC_SetPriority(INTADASGL_IRQn, 5);
75 * __NVIC_ClearPendingIRQ(INTADASGL_IRQn);
76 * __NVIC_EnableIRQ(INTADASGL_IRQn);
77 *
78 * CURRENTLY: DMA handles data movement. These interrupts are NOT enabled
79 * in the NVIC, so the CPU never enters the handlers below.
80 * The handlers are stubs for future use.
81 * ========================================================================== */

122void AINA_Init(void)
123{
124 /* GPIO ADC inputs (Unit A) */
125 PORT_L_Init();
126
127 /* [1] Enable ADC Unit A peripheral clock */
128 TSB_CG->FSYSMENB |= AINA_CG_FSYSMENB_IPMENB02;
129
130 /* [2] Disable ADC before configuration (ADEN must be 0 to change settings) */
131 TSB_ADA->CR0 &= ~ADxCR0_ADEN;
132
133 /* [3] Enable ADC conversion clock (ADCLK) */
134 TSB_CG->SPCLKEN |= CG_CGSPCLKEN_ADCKEN0;
135
136 /* [4] ADCLK prescaler = /4 -> SCLK = ADCLK/4. EXAZ left at reset (m=1) */
137 TSB_ADA->CLK &= ~ADxCLK_VADCLK_MASK;
138
139 /* [5] Enable ADC analog circuit, exit low-power mode */
140 TSB_ADA->MOD0 |= ADxMOD0_DACON;
141 TSB_ADA->MOD0 &= ~ADxMOD0_RCUT;
142
143 /* [6] Wait for analog stabilization (3 us minimum per datasheet) - Note3: 5.2.5. [ADxMOD0] (Mode Setting Register0)*/
144 SysTick_us(3U);
145
146 /* [7] Conversion timing — keep a DOCUMENTED MOD1 value (the 40 MHz recipe).
147 * Do NOT drop to reset 0x00004000: it is not a characterized operating
148 * config. At 2.5 MHz SCLK. */
149 TSB_ADA->MOD1 = ADxMOD1_N5_SGL;
150 TSB_ADA->MOD2 = 0;
151
152 /* [8] Select EXAZ0 sampling time for both channels */
153 TSB_ADA->EXAZSEL &= ~(ADxEXAZSEL_AINA15 | ADxEXAZSEL_AINA16);
154
155 /* [9] Program conversion sequence:
156 * TSET0: AINA16 -> REG0
157 * TSET1: AINA15 -> REG1 (ENINT set only in DMA builds)
158 */
159 TSB_ADA->TSET0 = (ADxTSETn_TRGSn_SGL | ADxTSETn_AINSTn_AINx16);
160 TSB_ADA->TSET1 = (ADxTSETn_TRGSn_SGL | ADxTSETn_AINSTn_AINx15 | ADC_TSET1_INT);
161
162 /* [10] DMA request on single conversion (DMA build) / none (polled) */
163 #ifdef ADC_USE_DMA
164 TSB_ADA->CR1 = ADxCR1_SGLDMEN;
165 #else
166 TSB_ADA->CR1 = 0U;
167 #endif
168
169 /* [11] Re-enable ADC to start operation */
170 TSB_ADA->CR0 |= ADxCR0_ADEN;
171}

177void AINC_Init(void)
178{
179 /* GPIO ADC inputs (Unit C) */
180 PORT_J_Init();
181
182 /* [1] Enable ADC Unit C peripheral clock */
183 TSB_CG->FSYSMENB |= AINC_CG_FSYSMENB_IPMENB04;
184
185 /* [2] Disable ADC before configuration */
186 TSB_ADC->CR0 &= ~ADxCR0_ADEN;
187
188 /* [3] Enable ADC conversion clock (ADCLK) */
189 TSB_CG->SPCLKEN |= CG_CGSPCLKEN_ADCKEN2;
190
191 /* [4] Set ADCLK prescaler = /4, sampling time for EXAZ0/EXAZ1 */
192 TSB_ADC->CLK &= ~ADxCLK_VADCLK_MASK;
193
194
195 /* [5] Enable ADC analog circuit, exit low-power mode */
196 TSB_ADC->MOD0 |= ADxMOD0_DACON;
197 TSB_ADC->MOD0 &= ~ADxMOD0_RCUT;
198
199 /* [6] Wait for analog stabilization */
200 SysTick_us(3U);
201
202 /* [7] Set conversion timing */
203 TSB_ADC->MOD1 = ADxMOD1_N5_SGL;
204 TSB_ADC->MOD2 = 0;
205
206 /* [8] Select EXAZ0 sampling time for both channels */
207 TSB_ADC->EXAZSEL &= ~(ADxEXAZSEL_AINC00 | ADxEXAZSEL_AINC01);
208
209 /* [9] Program conversion sequence:
210 * TSET0: AINC01 -> REG0
211 * TSET1: AINC00 -> REG1 (ENINT set only in DMA builds)
212 */
213 TSB_ADC->TSET0 = (ADxTSETn_TRGSn_SGL | ADxTSETn_AINSTn_AINx01);
214 TSB_ADC->TSET1 = (ADxTSETn_TRGSn_SGL | ADxTSETn_AINSTn_AINx00 | ADC_TSET1_INT);
215
216 /* [10] DMA request on single conversion (DMA build) / none (polled) */
217 #ifdef ADC_USE_DMA
218 TSB_ADC->CR1 = ADxCR1_SGLDMEN;
219 #else
220 TSB_ADC->CR1 = 0U;
221 #endif
222
223 /* [11] Re-enable ADC to start operation */
224 TSB_ADC->CR0 |= ADxCR0_ADEN;
225}

284uint16_t AINA_Read(uint8_t channel)
285{
286 uint16_t result = 0;
287
288 AINA_StartSGL();
289 while (TSB_ADA->ST & ADxST_SNGF) { ; }
290
291 if (channel == 16)

293 result = (uint16_t)((TSB_ADA->REG0 & ADxREGn_ADRn) >> 4U);

295 else if (channel == 15)

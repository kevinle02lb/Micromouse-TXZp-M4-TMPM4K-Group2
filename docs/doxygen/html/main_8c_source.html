107int main(void)
108{
109 SystemInit(); /* Initialize system clock and peripherals */
110 ModuleInit();
111
112
113 /* Control loop: fires once per 1 kHz tick. Order matters — each stage
114 * consumes the output of the one above it within the same tick. */
115 while (1)
116 {
117 if (Timebase_GetAndClear()) // true once per 1 ms tick
118 {
119 Encoder_Update(); // 1. read encoders, filter speed
120 Odometry_Update(); // 2. update pose from encoders
121 IR_SampleStep(); // 3. sample IR readings
122 Navigator_Update(); // 4. plan/turn/drive FSM step
123 Motion_Update(); // 5. PID -> motor PWM
124 }
125 }
126
127 return 0;
128}

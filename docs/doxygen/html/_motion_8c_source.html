62static void Motion_ApplyOutput(motor_t motor, float output)
63{
64 motor_dir_t dir;
65 float magnitude;
66
67 if (output > MOTION_DEADZONE)
68 {
69 dir = FORWARD;
70 magnitude = output;
71 }
72 else if (output < -MOTION_DEADZONE)
73 {
74 dir = REVERSE;
75 magnitude = -output;
76 }
77 else
78 {
79 dir = last_dir[motor];
80 magnitude = 0.0f;
81 }
82
83 last_dir[motor] = dir;
84
85 Motor_Set(motor, dir, (uint8_t)(magnitude + MOTION_ROUND_OFFSET));
86}

123void Motion_Update(void)
124{
125 float actual_left, actual_right;
126 float error_left, error_right;
127
128 if ((target_left == 0.0f) && (target_right == 0.0f))
129 {
130 PID_Reset(&pid_left);
131 PID_Reset(&pid_right);
132
133 output_left = 0.0f;
134 output_right = 0.0f;
135
136 Motion_ApplyOutput(MOTOR_LEFT, 0.0f);
137 Motion_ApplyOutput(MOTOR_RIGHT, 0.0f);
138 return;
139 }
140
141 actual_left = (float)Encoder_GetSpeed_CPS(MOTOR_LEFT);
142 actual_right = (float)Encoder_GetSpeed_CPS(MOTOR_RIGHT);
143
144 error_left = CalculateError(target_left, actual_left);
145 error_right = CalculateError(target_right, actual_right);
146
147 output_left = PID_Update(&pid_left, error_left);
148 output_right = PID_Update(&pid_right, error_right);
149
150 Motion_ApplyOutput(MOTOR_LEFT, (float)MOTOR_SIGN_LEFT * output_left);
151 Motion_ApplyOutput(MOTOR_RIGHT, (float)MOTOR_SIGN_RIGHT * output_right);
152}

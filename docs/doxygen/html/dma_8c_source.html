98void DMA_SetupForADC(void)
99{
100 /* Common ChnlCfg for both ADC units */
101 uint32_t chnl_cfg = DMA_DST_INC_HWORD /* dest: +2 bytes per transfer */
102 | DMA_DST_SIZE_HWORD /* dest: 16-bit writes */
103 | DMA_SRC_INC_WORD /* src: +4 bytes (REG0 → REG1) */
104 | DMA_SRC_SIZE_HWORD /* src: 16-bit reads (lower half of 32-bit reg) */
105 | DMA_R_POWER_2 /* arbitrate after 2 transfers */
106 | DMA_N_MINUS_1(2) /* 2 transfers total */
107 | DMA_NEXT_USEBURST /* bit 3 = 1: force burst */
108 | DMA_CYCLE_CTRL_CNT; /* cycle_ctrl = 010: Auto-request / Continuous Normal */
109
110 /* Channel 16: ADC Unit A → adc_a_buffer */
111 DMA_ConfigChannel(DMA_ADASLG_DMAREQ,
112 (uint32_t)&TSB_ADA->REG1,
113 (uint32_t)&adc_a_buffer[1],
114 chnl_cfg);
115
116 /* Channel 18: ADC Unit C → adc_c_buffer */
117 DMA_ConfigChannel(DMA_ADCSLG_DMAREQ,
118 (uint32_t)&TSB_ADC->REG1,
119 (uint32_t)&adc_c_buffer[1],
120 chnl_cfg);
121}

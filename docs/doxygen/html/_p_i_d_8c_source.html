95float PID_Update(pid_t *pid, float error)
96{
97 float proportional_term, integral_term, derivative_term, output;
98
99 // [1] Proportional
100 proportional_term = pid->kp * error;
101
102 // [2] Integral (accumulate, then clamp to desired range)
103 pid->integral += error * pid->dt;
104
105 if (pid->ki != 0.0f) // skip clamp math if I-term unused
106 {
107 float max_integral = pid->out_max / pid->ki;
108 float min_integral = pid->out_min / pid->ki;
109
110 if (pid->integral > max_integral)
111 pid->integral = max_integral;
112 else if (pid->integral < min_integral)
113 pid->integral = min_integral;
114 }
115
116 integral_term = pid->ki * pid->integral;
117
118 // [3] Derivative (rate of change of error)
119 pid->derivative = (error - pid->prev_error) / pid->dt;
120 derivative_term = pid->kd * pid->derivative;
121
122 // [4] Save error for next tick's derivative
123 pid->prev_error = error;
124
125 // [5] Sum and clamp
126 output = proportional_term + integral_term + derivative_term;
127
128 if (output > pid->out_max)
129 output = pid->out_max;
130 else if (output < pid->out_min)
131 output = pid->out_min;
132
133 return output;
134}

137static void IR_Acquire(uint16_t out[IR_COUNT])
138{
139#ifdef ADC_USE_DMA
140 volatile uint16_t *bufA;
141 volatile uint16_t *bufC;
142
143 Start_ADC(); /* Trigger both units, DMA moves results */
144 SysTick_us(5U); /* Wait for 2 conversions + DMA burst */
145
146 bufA = DMA_GetADCABuffer();
147 bufC = DMA_GetADCCBuffer();
148
149 /* DMA stores the raw register halfword — right-align to match polled path */
150 out[IR_FAR_LEFT] = (uint16_t)((bufA[0] & ADxREGn_ADRn) >> 4U); /* AINA16 */
151 out[IR_LEFT] = (uint16_t)((bufA[1] & ADxREGn_ADRn) >> 4U); /* AINA15 */
152 out[IR_RIGHT] = (uint16_t)((bufC[0] & ADxREGn_ADRn) >> 4U); /* AINC01 */
153 out[IR_FAR_RIGHT] = (uint16_t)((bufC[1] & ADxREGn_ADRn) >> 4U); /* AINC00 */
154#else
155 uint16_t a0, a1;
156 uint16_t c0, c1;
157
158 AINA_ReadPair(&a0, &a1); /* a0 = AINA16, a1 = AINA15 */
159 AINC_ReadPair(&c0, &c1); /* c0 = AINC01, c1 = AINC00 */
160
161 out[IR_FAR_LEFT] = a0;
162 out[IR_LEFT] = a1;
163 out[IR_RIGHT] = c0;
164 out[IR_FAR_RIGHT] = c1;
165#endif
166}

172static void IR_ComputeReflected(void)
173{
174 int i;
175
176 /* [1] Calculate reflected values (ambient cancellation) */
177 for (i = 0; i < IR_COUNT; i++)
178 {
179 if (ir_data.ambient[i] > ir_data.raw[i])
180 {
181 ir_data.reflected[i] = ir_data.ambient[i] - ir_data.raw[i];
182 }
183 else
184 {
185 ir_data.reflected[i] = 0U;
186 }
187 }
188
189 /* [2] IIR low-pass filter on reflected values */
190 for (i = 0; i < IR_COUNT; i++)
191 {
192 ir_data.filtered[i] = IR_FilterIIR(ir_data.filtered[i],
193 ir_data.reflected[i],
194 IR_FILTER_SHIFT);
195 }
196}

203static void IR_UpdateWalls(void)
204{

206

212 wall_state[i] = (distance < (IR_WALL_DISTANCE_MM + IR_WALL_HYSTERESIS_MM));
213 else
214 wall_state[i] = (distance < (IR_WALL_DISTANCE_MM - IR_WALL_HYSTERESIS_MM));
215
216 ir_data.wallDetected[i] = wall_state[i];
217 }

236void IR_SampleAll(void)
237{
238 /* [1] Ambient reading — all emitters OFF */
239 IR_AllEmittersOff();
240 SysTick_us(20U);
241 IR_Acquire(ir_data.ambient);
242
243 /* [2] Reflected reading — all emitters ON */
244 IR_AllEmittersOn();
245 SysTick_us(600U);
246 IR_Acquire(ir_data.raw);
247
248 /* [3] Ambient cancel + IIR filter */
249 IR_ComputeReflected();
250
251 /* [4] Power save — turn off emitters between samples */
252 IR_AllEmittersOff();
253
254 /* [5] Update wall & flags from filtered data */
255 IR_UpdateWalls();
256}

267void IR_SampleStep(void)
268{
269 switch (ir_phase)
270 {
271 case IR_PHASE_IDLE:
272 default:
273 IR_AllEmittersOff();
274 ir_phase = IR_PHASE_AMBIENT_SETTLE;
275 break;
276
277 case IR_PHASE_AMBIENT_SETTLE:
278 /* [1] Emitters off — prior reflection decays this tick */
279 ir_phase = IR_PHASE_AMBIENT_READ;
280 break;
281
282 case IR_PHASE_AMBIENT_READ:
283 /* [2] Ambient reading — all emitters OFF */
284 IR_Acquire(ir_data.ambient);
285 ir_phase = IR_PHASE_EMITTER_ON;
286 break;
287
288 case IR_PHASE_EMITTER_ON:
289 /* [3] Emitters ON — arm settle counter */
290 IR_AllEmittersOn();
291 ir_settle_cnt = IR_SETTLE_TICKS;
292 ir_phase = IR_PHASE_SETTLE;
293 break;
294
295 case IR_PHASE_SETTLE:
296 /* [4] Hold on — receiver RC settles across ticks */
297 if (--ir_settle_cnt == 0U)
298 {
299 ir_phase = IR_PHASE_READ;
300 }
301 break;
302
303 case IR_PHASE_READ:
304 /* [5] Reflected reading, then power save (Turn off emitters) */
305 IR_Acquire(ir_data.raw);
306 IR_AllEmittersOff();
307 ir_phase = IR_PHASE_COMPUTE;
308 break;
309
310 case IR_PHASE_COMPUTE:
311 /* [6] Ambient cancel + filter + walls */
312 IR_ComputeReflected();
313 IR_UpdateWalls();
314 ir_phase = IR_PHASE_AMBIENT_SETTLE;
315 break;
316 }
317}

368float IR_GetDistance_mm(ir_channel_t ch)
369{
370 const ir_cal_point_t *cal = ir_cal[ch];
371 uint16_t value = ir_data.filtered[ch];
372 uint8_t i;
373 float frac;
374
375 if (value >= cal[0].adc)
376 return (float)cal[0].mm;
377
378 for (i = 0U; i < (IR_CAL_POINTS - 1U); i++)
379 {
380 if (value >= cal[i + 1U].adc)
381 {
382 frac = (float)(cal[i].adc - value) /
383 (float)(cal[i].adc - cal[i + 1U].adc);
384
385 return (float)cal[i].mm +
386 (frac * (float)(cal[i + 1U].mm - cal[i].mm));
387 }
388 }
389
390 return (float)cal[IR_CAL_POINTS - 1U].mm;
391}

494uint16_t IR_FilterIIR(uint16_t prev, uint16_t curr, uint8_t shift)
495{
496 uint16_t delta;
497 uint16_t result;
498
499 if (curr > prev)
500 {
501 delta = curr - prev;
502 result = prev + (delta >> shift);
503 }
504 else
505 {
506 delta = prev - curr;
507 result = prev - (delta >> shift);
508 }
509
510 return result;
511}

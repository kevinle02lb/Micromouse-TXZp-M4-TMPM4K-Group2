117void UART_Init(void)
118{
119 /* [1] Enable clock supply: Port C + UART ch0 */
120 TSB_CG->FSYSMENA |= CG_PORTC; /* Port C Clk En */
121 TSB_CG->FSYSMENA |= CG_FSYSMENA_IPMENA21; /* UART CH0 Clk En */
122
123 /* [2] Route PC0 (UT0RXD) / PC1 (UT0TXDA) to the UART function & Reset UART */
124 UART_PORT_C_Init();
125 TSB_UART0->SWRST = 0x02U; /* Software Reset */
126 TSB_UART0->SWRST = 0x01U;
127 while ((TSB_UART0->SWRST & UARTx_SWRST_SWRSTF) != 0U) { ; } /* wait SWRSTF=0 */
128
129 /* [3] Wait until the setting registers are writable (<SUE> = 0) */
130 while ( (TSB_UART0->SR & UARTx_SR_SUE) != 0U ) { ; }
131
132 /* [4] Prescaler: 1/1 */
133 TSB_UART0->CLK &= ~UARTx_CLK_PRSEL_MASK;
134
135 /* [5] Frame format: 8-bit, no parity, 1 STOP bit, LSB first */
136 TSB_UART0->CR0 = UARTx_CR0_SM_8BIT << CR0_SM_POS;
137
138 /* [6] Baud rate: KEN = 1, N = 43, K = 38 -> 115191 Bd */
139 TSB_UART0->BRD = UARTx_BRD_KEN_MASK | (UARTx_BRD_BRN << BRD_BRN_POS) | (UARTx_BRD_BRK << BRD_BRK_POS);
140
141 /* [7] Enable transmission (normal operation, no trigger) */
142 TSB_UART0->TRANS |= UARTx_TRANS_TXE_MASK;
143}

177void UART_SendUint(uint16_t val)
178{
179 char buf[6]; /* Max "65535" + NUL terminator */
180 int i = (int)(sizeof(buf) - 1U);
181 buf[i] = '\0';
182
183 if ( val == 0U )
184 {
185 UART_SendByte('0');
186 return;
187 }
188
189 /* Extract digits backwards, least significant first */
190 while ( val > 0U )
191 {
192 buf[--i] = (char)('0' + (val % 10U));
193 val /= 10U;
194 }
195
196 UART_SendString(&buf[i]);
197}

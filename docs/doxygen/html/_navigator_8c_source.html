154static float SegmentDistance_mm(void)
155{
156 return 0.5f * (SegmentLeft_mm() + SegmentRight_mm());

168static float SegmentArc_mm(void)
169{
170 return 0.5f * (SegmentRight_mm() - SegmentLeft_mm()) * turn_sign;

188static float StraightnessTrim(void)
189{
190 float skew = SegmentRight_mm() - SegmentLeft_mm();
191 float error = 0.0f;
192 bool left = IR_IsWallPresent(IR_FAR_LEFT);
193 bool right = IR_IsWallPresent(IR_FAR_RIGHT);
194
195 /* Referenced Displacement on Right */
196 if (left && right)
197 error = IR_GetDistance_mm(IR_FAR_RIGHT) - IR_GetDistance_mm(IR_FAR_LEFT);
198 else if (left)
199 error = 2.0f * (IR_SIDE_TARGET_MM - IR_GetDistance_mm(IR_FAR_LEFT));
200 else if (right)
201 error = 2.0f * (IR_GetDistance_mm(IR_FAR_RIGHT) - IR_SIDE_TARGET_MM);
202
203 return (KP_WALL * error) + (KP_STRAIGHT * skew);
204}

223static void ApplyFrontWallCorrection(float traveled_mm)
224{
225 float front_mm;
226
227 if (!IR_IsWallPresent(IR_LEFT) || !IR_IsWallPresent(IR_RIGHT))
228 return;
229
230 front_mm = 0.5f * (IR_GetDistance_mm(IR_LEFT) + IR_GetDistance_mm(IR_RIGHT));
231
232 segment.distance_mm = traveled_mm + (front_mm - IR_FRONT_TARGET_MM);
233}

295void Navigator_Update(void)
296{
297 switch (nav_state)
298 {
299 /* ==============================================================
300 * NAV_PLAN — read walls, ask FloodFill, arm the first segment
301 * ============================================================== */
302 case NAV_PLAN:
303 {
304 CommandWheels(0.0f, 0.0f);
305
306 FloodFill_SetWalls(IR_IsWallPresent(IR_LEFT) &&
307 IR_IsWallPresent(IR_RIGHT),
308 IR_IsWallPresent(IR_FAR_LEFT),
309 IR_IsWallPresent(IR_FAR_RIGHT));
310
311 current_action = FloodFill_Plan();
312
313 switch (current_action)
314 {
315 case FLOODFILL_FORWARD:
316 BeginDrive(CELL_SIZE_MM);
317 break;
318
319 case FLOODFILL_TURN_LEFT: // CCW
320 BeginTurn(M_PI_DIV_2);
321 break;
322
323 case FLOODFILL_TURN_RIGHT: // CW
324 BeginTurn(-M_PI_DIV_2);
325 break;
326
327 case FLOODFILL_TURN_AROUND:
328 BeginTurn(M_PI);
329 break;
330
331 case FLOODFILL_STOP:
332 default:
333 Motion_Stop();
334 nav_state = NAV_FINISHED;
335 break;
336 }
337 break;
338 }
339
340 /* ==============================================================
341 * NAV_TURN — pivot until the commanded arc is covered
342 * ============================================================== */
343 case NAV_TURN:
344 {
345 float swept = SegmentArc_mm();
346 float v = Profile_Step(&segment, swept);
347
348 CommandWheels(-v * turn_sign, v * turn_sign);
349
350 if (Profile_IsComplete(&segment, swept))
351 {
352 CommandWheels(0.0f, 0.0f);
353 BeginDrive(CELL_SIZE_MM);
354 }
355 break;
356 }
357
358 /* ==============================================================
359 * NAV_DRIVE — advance one cell under the straightness trim
360 * ============================================================== */
361 case NAV_DRIVE:
362 {
363 float traveled = SegmentDistance_mm();
364 float v;
365 float trim;
366
367 ApplyFrontWallCorrection(traveled);
368
369 v = Profile_Step(&segment, traveled);
370 trim = StraightnessTrim();
371
372 CommandWheels(v + trim, v - trim);
373
374 if (Profile_IsComplete(&segment, traveled))
375 {
376 CommandWheels(0.0f, 0.0f);
377 settle_ticks = 0U;
378 nav_state = NAV_SETTLE;
379 }
380 break;
381 }
382
383 /* ==============================================================
384 * NAV_SETTLE - hold at the cell centre, then commit the move
385 * ============================================================== */
386 case NAV_SETTLE:
387 {
388 CommandWheels(0.0f, 0.0f);
389 settle_ticks++;
390
391 if (settle_ticks >= SETTLE_TICKS)
392 {
393 FloodFill_ReportDone(current_action);
394 nav_state = NAV_PLAN;
395 }
396 break;
397 }
398
399 /* ==============================================================
400 * NAV_FINISHED — terminal, bridges left in standby
401 * ============================================================== */
402 case NAV_FINISHED:
403 default:
404 break;
405 }
406}

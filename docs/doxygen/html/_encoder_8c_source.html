23/* ==========================================================================
24 * PHYSICAL CALCULATIONS & UNIT CONVERSIONS
25 * ==========================================================================
26 *
27 * From Pololu Micro Metal Gearmotor Datasheet (Rev 6-2, Page 2):
28 *
29 * "The two-channel Hall effect encoder senses the rotation of a 6-pole
30 * magnetic disc on a rear protrusion of the motor shaft, providing a
31 * resolution of 12 counts per revolution (CPR) of the motor shaft when
32 * counting both edges of both channels."
33 *
34 * "To compute the gearbox output CPR, multiply 12 by the gearbox
35 * reduction factor."
36 *
37 * Gearbox ratio: 30:1 (nominal)
38 * Exact gear ratio: 29.89:1 (from Page 3 gearbox table)
39 * Wheel CPR = 12 × 29.89 = 358.68 ≈ 359 counts per wheel revolution.
40 *
41 * For simplicity in code constants, use 360 counts per wheel revolution.
42 *
43 * Control Loop Tick: 1 kHz (1 ms period)
44 * - Encoder_Update() is called once per tick.
45 * - delta = counts detected in the last 1 ms.
46 * - speed_raw (CPS) = delta × 1000 (1 second = 1000 ms)
47 *
48 * Example:
49 * - delta = 3 counts in 1 ms → CPS = 3000
50 * - 3000 CPS / 360 counts/rev = 8.33 wheel revolutions per second.
51 */

82static int32_t Encoder_WindowedSpeed(encoder_t *enc)
83{
84 uint8_t oldest_head;
85 int32_t counts_in_window;
86
87 enc->history_head = (uint8_t)((enc->history_head + 1U) % (SPEED_WINDOW_TICKS + 1U));
88 enc->position_history[enc->history_head] = enc->position;
89
90 oldest_head = (uint8_t)((enc->history_head + 1U) % (SPEED_WINDOW_TICKS + 1U));
91 counts_in_window = enc->position - enc->position_history[oldest_head];
92
93 return counts_in_window * (1000 / SPEED_WINDOW_TICKS);
94}

106void Encoder_Init(void)
107{
108 /* Initialise the hardware drivers */
109 ENC32A_Init();
110
111 /* Clear all state */
112 for (uint8_t i = 0; i < 2; i++)
113 {
114 enc_state[i].raw_count = 0;
115 enc_state[i].prev_count = 0;
116 enc_state[i].delta = 0;
117 enc_state[i].position = 0;
118 enc_state[i].speed = 0;
119
120 /* Speed window */
121 enc_state[i].history_head = 0;
122 for (uint8_t j = 0; j <= SPEED_WINDOW_TICKS; j++)
123 enc_state[i].position_history[j] = 0;
124 }
125
126
127
128 /* Read the current counter values to set prev_count */
129 enc_state[MOTOR_LEFT].prev_count = ENC0_ReadCount();
130 enc_state[MOTOR_RIGHT].prev_count = ENC2_ReadCount();
131 enc_state[MOTOR_LEFT].raw_count = enc_state[MOTOR_LEFT].prev_count;
132 enc_state[MOTOR_RIGHT].raw_count = enc_state[MOTOR_RIGHT].prev_count;
133}

141void Encoder_Update(void)
142{
143 int32_t raw, delta;
144
145 /* Left motor */
146 raw = ENC0_ReadCount();
147 delta = raw - enc_state[MOTOR_LEFT].prev_count;
148 enc_state[MOTOR_LEFT].raw_count = raw;
149 enc_state[MOTOR_LEFT].delta = delta;
150 enc_state[MOTOR_LEFT].position += delta;
151 enc_state[MOTOR_LEFT].speed = Encoder_WindowedSpeed(&enc_state[MOTOR_LEFT]);
152 enc_state[MOTOR_LEFT].prev_count = raw;
153
154 /* Right motor */
155 raw = ENC2_ReadCount();
156 delta = raw - enc_state[MOTOR_RIGHT].prev_count;
157 enc_state[MOTOR_RIGHT].raw_count = raw;
158 enc_state[MOTOR_RIGHT].delta = delta;
159 enc_state[MOTOR_RIGHT].position += delta;
160 enc_state[MOTOR_RIGHT].speed = Encoder_WindowedSpeed(&enc_state[MOTOR_RIGHT]);
161 enc_state[MOTOR_RIGHT].prev_count = raw;
162}

47void PORT_A_Init(void)
48{
49 TSB_CG->FSYSMENA |= CG_PORTA;
50
51 /* PA3 = T32A00OUTA, PA4 = T32A00OUTB — PWM outputs */
52 PA_FRn_Clear(Px3_MASK | Px4_MASK);
53 TSB_PA->FR4 |= Px3_MASK | Px4_MASK; /* FR4 = T32A00 function */
54 TSB_PA->CR |= Px3_MASK | Px4_MASK;
55}

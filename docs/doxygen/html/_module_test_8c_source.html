182 {

186

363 {
364 if (m & E_FL) { GPIO_U_SetData(Px1_MASK); } else { GPIO_U_ClrData(Px1_MASK); }

636/* ==========================================================================
637 * Turn Test
638 *
639 * One in-place turn driven the same way the Navigator drives one: a single
640 * trapezoidal Profile segment on the arc each wheel traces.
641 *
642 * A pivot of theta radians gives each wheel an arc of
643 *
644 * arc = theta * WHEELBASE_MM / 2
645 *
646 * Streaming continues through the hold, which is where overshoot appears.
647 *
648 * Speed constants are local to this block. A test harness reaching into
649 * another module's tuning would couple the two.
650 *
651 * Columns: ms,arc,deg,pvL,pvR,sp,state
652 * ms milliseconds since the turn was commanded
653 * arc wheel arc covered, mm
654 * deg odometry heading, degrees
655 * pvL/R measured wheel speed, mm/s
656 * sp profile setpoint, mm/s
657 * state 0 turning, 1 holding, 2 done
658 * ========================================================================== */
659#ifdef TURN_TEST
660
661 #define TURN_TEST_ANGLE_RAD M_PI_DIV_2 /* magnitude, always positive */
662 #define TURN_TEST_CCW 1 /* 1 = left/CCW, 0 = right/CW */
663
664 #define TURN_TEST_SPEED_MM_S 300.0f /* cruise ceiling, per wheel */
665 #define TURN_TEST_MIN_MM_S 0.0f /* floor, clears stiction */
666 #define TURN_TEST_ACCEL_MM_S2 500.0f /* ramp and brake */
667
668 #define TURN_TEST_HOLD_MS 500U /* stationary hold after the turn */
669 #define TURN_TEST_PRINT_EVERY 10U /* 1 kHz / 10 = 100 Hz */
670
676 static float TurnTest_Speed_mm_s(motor_t motor)
677 {
678 return (float)Encoder_GetSpeed_CPS(motor) * MM_PER_COUNT;
679 }
680
687 static void TurnTest_Command(float left_mm_s, float right_mm_s)
688 {
689 Motion_SetSpeed(left_mm_s / MM_PER_COUNT, right_mm_s / MM_PER_COUNT);
690 }
691
696 static void TurnTest_Run(void)
697 {
698 profile_t seg;
699 int32_t startL, startR;
700 float arc_target_mm;
701 float sign;
702 float swept = 0.0f;
703 float v = 0.0f;
704 uint32_t ms = 0U;
705 uint32_t tick = 0U;
706 uint32_t hold = 0U;
707 uint8_t state = 0U;
708
709 sign = (TURN_TEST_CCW) ? 1.0f : -1.0f;
710 arc_target_mm = TURN_TEST_ANGLE_RAD * WHEELBASE_MM * 0.5f;
711
712 UART_SendString("target,");
713 Test_SendDecimal(arc_target_mm);
714 UART_CRLF();
715 UART_SendString("ms,arc,deg,pvL,pvR,sp,state");
716 UART_CRLF();
717
718 Odometry_Reset();
719
720 startL = Encoder_GetPosition(MOTOR_LEFT);
721 startR = Encoder_GetPosition(MOTOR_RIGHT);
722
723 Profile_Begin(&seg, arc_target_mm,
724 TURN_TEST_SPEED_MM_S,
725 TURN_TEST_MIN_MM_S,
726 TURN_TEST_ACCEL_MM_S2);
727
728 while (1)
729 {
730 if (!Timebase_GetAndClear())
731 continue;
732
733 Encoder_Update();
734 Odometry_Update();
735
736 /* Wheels sweep equal and opposite arcs, so each covers half the
737 difference. The sign makes it count up for either direction. */
738 swept = 0.5f * sign * MM_PER_COUNT *
739 ((float)(Encoder_GetPosition(MOTOR_RIGHT) - startR) -
740 (float)(Encoder_GetPosition(MOTOR_LEFT) - startL));
741
742 switch (state)
743 {
744 case 0U:
745 v = Profile_Step(&seg, swept);
746 TurnTest_Command(-v * sign, v * sign);
747
748 if (Profile_IsComplete(&seg, swept))
749 {
750 hold = 0U;
751 state = 1U;
752 }
753 break;
754
755 case 1U:
756 v = 0.0f;
757 TurnTest_Command(0.0f, 0.0f);
758
759 if (++hold >= TURN_TEST_HOLD_MS)
760 state = 2U;
761 break;
762
763 default:
764 v = 0.0f;
765 TurnTest_Command(0.0f, 0.0f);

791/* ==========================================================================
792 * Drive Test
793 *
794 * One straight move of a fixed distance, then a hold. Compare the printed
795 * dist10 against a ruler on the floor.
796 *
797 * Columns: ms,dist,encL,encR,sp,state
798 * ms milliseconds since the move was commanded
799 * dist distance covered, mm
800 * encL left encoder counts since the move started
801 * encR right encoder counts since the move started
802 * sp profile setpoint, mm/s
803 * state 0 driving, 1 done
804 * ========================================================================== */
805#ifdef DRIVE_TEST
806
807 #define DRIVE_TEST_DISTANCE_MM 180.0f
808 #define DRIVE_TEST_SPEED_MM_S 300.0f
809 #define DRIVE_TEST_MIN_MM_S 0.0f
810 #define DRIVE_TEST_ACCEL_MM_S2 300.0f
811 #define DRIVE_TEST_PRINT_EVERY 10U
812
813 static void DriveTest_Run(void)
814 {
815 profile_t seg;
816 int32_t startL, startR, dL, dR;
817 float dist = 0.0f;
818 float v = 0.0f;
819 uint32_t ms = 0U;
820 uint32_t tick = 0U;
821 uint8_t state = 0U;
822
823 UART_SendString("target,");
824 Test_SendDecimal(DRIVE_TEST_DISTANCE_MM);
825 UART_CRLF();
826 UART_SendString("ms,dist,encL,encR,sp,state");
827 UART_CRLF();
828
829 startL = Encoder_GetPosition(MOTOR_LEFT);
830 startR = Encoder_GetPosition(MOTOR_RIGHT);
831
832 Profile_Begin(&seg, DRIVE_TEST_DISTANCE_MM,
833 DRIVE_TEST_SPEED_MM_S, DRIVE_TEST_MIN_MM_S,
834 DRIVE_TEST_ACCEL_MM_S2);
835
836 while (1)
837 {
838 if (!Timebase_GetAndClear())
839 continue;
840
841 Encoder_Update();
842
843 dL = Encoder_GetPosition(MOTOR_LEFT) - startL;
844 dR = Encoder_GetPosition(MOTOR_RIGHT) - startR;
845 dist = 0.5f * (float)(dL + dR) * MM_PER_COUNT;
846
847 if (state == 0U)
848 {
849 v = Profile_Step(&seg, dist);
850
851 if (Profile_IsComplete(&seg, dist))
852 {
853 v = 0.0f;
854 state = 1U;
855 }
856 }
857
858 Motion_SetSpeed(v / MM_PER_COUNT, v / MM_PER_COUNT);
859 Motion_Update();
860 ms++;
861
862 if (++tick < DRIVE_TEST_PRINT_EVERY)
863 continue;
864
865 tick = 0U;
866
867 UART_SendInt((int32_t)ms); UART_SendByte(',');
868 Test_SendDecimal(dist); UART_SendByte(',');
869 UART_SendInt(dL); UART_SendByte(',');
870 UART_SendInt(dR); UART_SendByte(',');
871 Test_SendDecimal(v); UART_SendByte(',');
872 UART_SendInt((int32_t)state);
873 UART_CRLF();
874 }
875 }

877#endif /* DRIVE_TEST */

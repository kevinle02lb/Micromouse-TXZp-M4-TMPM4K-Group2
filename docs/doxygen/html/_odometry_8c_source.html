91void Odometry_Update(void)
92{
93 // Step 1: Calculate wheel displacements
94 int32_t left_pos = Encoder_GetPosition(MOTOR_LEFT);
95 int32_t right_pos = Encoder_GetPosition(MOTOR_RIGHT);
96
97 int32_t left_delta = left_pos - odo.prev_left_pos;
98 int32_t right_delta = right_pos - odo.prev_right_pos;
99
100 float D_left = (float)left_delta * MM_PER_COUNT;
101 float D_right = (float)right_delta * MM_PER_COUNT;
102
103 // Step 2: Calculate average displacement and orientation change
104 float D_avg = (D_left + D_right) / AVG_DIVISOR;
105 float delta_theta = (D_right - D_left) / WHEELBASE_MM;
106
107 // Step 3: Calculate changes in global position
108 float theta_new = odo.heading_rad + delta_theta;
109
110 float delta_x = D_avg * cosf(theta_new);
111 float delta_y = D_avg * sinf(theta_new);
112
113 // Step 4: Update global position and orientation
114 odo.x_mm += delta_x;
115 odo.y_mm += delta_y;
116 odo.heading_rad = theta_new;
117
118 // Normalize heading to [-PI, PI]
119 if (odo.heading_rad > M_PI)
120 odo.heading_rad -= M_2PI;
121 else if (odo.heading_rad < -M_PI)
122 odo.heading_rad += M_2PI;
123
124 // Store encoder positions for next tick
125 odo.prev_left_pos = left_pos;
126 odo.prev_right_pos = right_pos;
127
128 // Accumulate total wheel distances (optional, for diagnostics)
129 odo.left_dist_mm += D_left;
130 odo.right_dist_mm += D_right;
131}

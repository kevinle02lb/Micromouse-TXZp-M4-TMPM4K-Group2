49void ENC0_Init(void)
50{
51 PORT_N_Init();
52
53 /* [1] Enable peripheral clock */
54 TSB_CG->FSYSMENB |= ENC0_CG_FSYSMENB_IPMENB06;
55
56 /* [2] Stop encoder before config (ENRUN = 0) */
57 TSB_EN0->TNCR &= ~TNCR_ENRUN_MASK;
58
59 /* [3] Operation mode: Encoder (000), no Z, no Phase 3 */
60 TSB_EN0->TNCR &= ~(TNCR_MODE_MASK | TNCR_P3EN_MASK | TNCR_ZEN_MASK);
61 /* Clear counter */
62 TSB_EN0->TNCR |= TNCR_ENCLR_MASK;
63 TSB_EN0->TNCR &= ~TNCR_ENCLR_MASK;
64
65
66 /* [4] Input circuit: 10 MHz sample clock, 400 ns noise cancellation */
67 TSB_EN0->CLKCR = CLKCR_SPLCKS_8DIV; /* Fsys/8 = 10 MHz */
68 TSB_EN0->INPCR &= ~(INPCR_SYNCSPLEN_MASK | INPCR_SYNCSPLMD_MASK | INPCR_SYNCNCZEN_MASK);
69 TSB_EN0->INPCR = (TSB_EN0->INPCR & ~INPCR_NCT_MASK) | INPCR_NCT_4CYC;
70
71 /* [5] Decoder: CW/CCW edge detection (default) */
72 TSB_EN0->TNCR &= ~TNCR_DECMD_MASK;
73
74 /* [6] Counter: max reload value */
75 TSB_EN0->RELOAD = RELOAD_MAX;
76
77 /* [7] Interrupts: disabled */
78 TSB_EN0->INT = 0;
79 TSB_EN0->INTCR &= ~INTCR_CLEAR_ALL;
80
81 ENC0_Start();
82}

84void ENC2_Init(void)
85{

91 /* [2] Stop encoder before config */

94 /* [3] Operation mode: Encoder, no Z, no Phase 3 */
95 TSB_EN2->TNCR &= ~(TNCR_MODE_MASK | TNCR_P3EN_MASK | TNCR_ZEN_MASK);
96 /* Clear counter */
97 TSB_EN2->TNCR |= TNCR_ENCLR_MASK;
98 TSB_EN2->TNCR &= ~TNCR_ENCLR_MASK;
99
100

102 TSB_EN2->CLKCR = CLKCR_SPLCKS_8DIV;
103 TSB_EN2->INPCR &= ~(INPCR_SYNCSPLEN_MASK | INPCR_SYNCSPLMD_MASK | INPCR_SYNCNCZEN_MASK);
104 TSB_EN2->INPCR = (TSB_EN2->INPCR & ~INPCR_NCT_MASK) | INPCR_NCT_4CYC;
105
106 /* [5] Decoder: CW/CCW edge detection */
107 TSB_EN2->TNCR &= ~TNCR_DECMD_MASK;
108
109 /* [6] Counter: max reload */
110 TSB_EN2->RELOAD = RELOAD_MAX;
